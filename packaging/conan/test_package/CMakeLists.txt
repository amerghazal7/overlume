cmake_minimum_required(VERSION 3.16)
project(overlume_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
# vcpkg consumers of the static product pass -DOVERLUME_STATIC=ON; Conan ignores it (CMakeDeps selects the target).
option(OVERLUME_STATIC "vcpkg: find the static component" OFF)
if(OVERLUME_STATIC)
  find_package(overlume CONFIG REQUIRED COMPONENTS static)
else()
  find_package(overlume CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
if(TARGET overlume::overlume)
  target_link_libraries(consumer PRIVATE overlume::overlume)
else()
  target_link_libraries(consumer PRIVATE overlume::overlume_static)
endif()
